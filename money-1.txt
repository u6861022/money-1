    <!-- ⚙️ 核心邏輯 JavaScript -->
    <script>
        const GAS_URL = "https://google.com";
        let allRecords = [];
        let expenseChart = null;

        // 頁面初始化
        window.onload = function() {
            // 1. 初始化新增表單的日期為今天
            document.getElementById('inputDate').value = new Date().toISOString().split('T')[0];
            
            // 2. 初始化年份下拉選單（前後2年）
            const currentYear = new Date().getFullYear();
            const yearSelect = document.getElementById('filterYear');
            for(let y = currentYear - 2; y <= currentYear + 1; y++) {
                let opt = document.createElement('option');
                opt.value = y.toString();
                opt.textContent = y + " 年";
                if(y === currentYear) opt.selected = true;
                yearSelect.appendChild(opt);
            }

            // 3. 預設選中當前月份
            const currentMonth = (new Date().getMonth() + 1).toString().padStart(2, '0');
            document.getElementById('filterMonth').value = currentMonth;

            // 4. 監聽篩選選單的切換事件
            document.getElementById('filterYear').addEventListener('change', filterAndRender);
            document.getElementById('filterMonth').addEventListener('change', filterAndRender);

            // 5. 先讀取本地緩存，確保立即有畫面
            const cache = localStorage.getItem('records');
            if (cache) {
                allRecords = JSON.parse(cache);
                filterAndRender();
            }

            // 6. 從雲端獲取最新同步資料
            fetch(GAS_URL)
            .then(res => res.json())
            .then(data => {
                if (data) {
                    allRecords = data;
                    localStorage.setItem('records', JSON.stringify(allRecords));
                    document.getElementById('syncStatus').textContent = "🟢 雲端同步完成";
                    document.getElementById('syncStatus').className = "text-sm text-emerald-500 font-medium mt-1";
                    filterAndRender();
                }
            })
            .catch(err => {
                console.error("雲端讀取失敗，採用本地離線快取資料:", err);
                document.getElementById('syncStatus').textContent = "⚠️ 離線瀏覽中";
                document.getElementById('syncStatus').className = "text-sm text-rose-400 font-medium mt-1";
            });
        };

        // 核心功能：根據下拉選單過濾並渲染畫面與圖表
        function filterAndRender() {
            const targetYear = document.getElementById('filterYear').value;
            const targetMonth = document.getElementById('filterMonth').value;

            // 過濾符合年月份的資料
            const filtered = allRecords.filter(item => {
                if (!item.date) return false;
const recordYear = item.date.substring(0, 4);
const recordMonth = item.date.substring(5, 7);
if (targetMonth === "all") {
return recordYear === targetYear;
} else {
return recordYear === targetYear && recordMonth === targetMonth;
}
});
// 計算統計金額
let total = 0;
let categoryTotals = { "日常伙食": 0, "交通娛樂": 0, "生活雜支": 0, "固定開銷": 0, "其他支出": 0 };
filtered.forEach(item => {
const amt = Number(item.amount) || 0;
total += amt;
if (categoryTotals[item.type] !== undefined) {
categoryTotals[item.type] += amt;
}
});
// 更新統計卡片
document.getElementById('totalExpense').textContent = $${total.toLocaleString()};
document.getElementById('totalCount').textContent = filtered.length;
document.getElementById('avgExpense').textContent = filtered.length > 0 ? $${Math.round(total / filtered.length).toLocaleString()} : "$0";
// 渲染明細表格
const tbody = document.getElementById('recordTableBody');
tbody.innerHTML = "";
filtered.sort((a,b) => b.date.localeCompare(a.date)).forEach(item => {
const tr = document.createElement('tr');
tr.innerHTML = `
${item.date}
${item.type}
$${Number(item.amount).toLocaleString()}
${item.note || '-'}
`;
tbody.appendChild(tr);
});
// 渲染圓餅圖
updateChart(categoryTotals);
}
// 繪製或更新 Chart.js 圓餅圖
function updateChart(categoryTotals) {
const ctx = document.getElementById('expenseChart').getContext('2d');
const dataValues = Object.values(categoryTotals);
if (expenseChart) {
expenseChart.data.datasets[0].data = dataValues;
expenseChart.update();
} else {
expenseChart = new Chart(ctx, {
type: 'doughnut',
data: {
labels: Object.keys(categoryTotals),
datasets: [{
data: dataValues,
backgroundColor: ['#ef4444', '#3b82f6', '#f59e0b', '#10b981', '#64748b'],
borderWidth: 2,
borderColor: '#ffffff'
}]
},
options: {
plugins: { legend: { display: false } },
cutout: '65%'
}
});
}
}
// 新增記錄
function addRecord() {
const date = document.getElementById('inputDate').value;
const type = document.getElementById('inputType').value;
const amount = document.getElementById('inputAmount').value;
const note = document.getElementById('inputNote').value;
if (!date || !amount || amount <= 0) {
alert("請填寫正確的日期和金額！");
return;
}
const newRecord = { date, type, amount: Number(amount), note };
// 預先塞入前端陣列並更新畫面（體驗較流暢）
allRecords.push(newRecord);
localStorage.setItem('records', JSON.stringify(allRecords));
filterAndRender();
// 清空輸入欄位
document.getElementById('inputAmount').value = "";
document.getElementById('inputNote').value = "";
// 非同步上傳至 Google 試算表
fetch(GAS_URL, {
method: "POST",
mode: "no-cors",
headers: { "Content-Type": "application/json" },
body: JSON.stringify(newRecord)
})
.then(() => {
console.log("資料成功同步到 Google Sheets");
})
.catch(err => console.error("雲端背景同步失敗:", err));
}
